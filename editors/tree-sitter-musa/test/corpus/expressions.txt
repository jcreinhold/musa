piece "Expressions" {
  let choices: List((Pitch, Option(Pitch))) = [(c4, Some(e4)), (g4, None)];
  fn choose(value: Option(Pitch), fallback: Pitch) -> Pitch {
    match value {
      None -> fallback,
      Some(found) -> found,
    }
  }
  let raised: Pitch = transform(fn (from: Pitch) -> Pitch { choose(Some(from), c4) }, e4);
  fn melody(root: Pitch) -> EventTrack(WrittenTime, ScoreFact) {
    music {
      root/4
      use answer(root);
    }
  }
  score { part piano { voice one { c4/1 } } }
}

fn twice(x: Text) -> Text {
  let doubled: Text = text_join([x, x]);
  text_join([doubled, doubled])
}

fn stacked(x: Text) -> Text {
  let one = text_join([x, "a"]);
  let two = text_join([one, "b"]);
  text_join([one, two])
}

fn in_an_arm(written: Bool) -> Text {
  match written {
    True -> let word = "yes"; text_join([word, word]),
    False -> "no",
  }
}
