library {
  data Shape {
    Silence,
    Sounded(sounded: Pitch, held: Duration),
    Then(first: Shape, second: Shape),
  }

  data Pair<A, B> {
    Both(left: A, right: B),
  }

  fn describe(shape: Shape) -> Nat {
    match shape {
      Silence -> 0,
      Sounded(heard, lasting) -> 1,
      Then(before, after) -> 2,
    }
  }

  fn left_of(both: Pair<Nat, Nat>) -> Nat { both.left }
}

library {
  data Dynamic {
    Quiet,
    Loud(level: Nat),
  }

  data Wrapper {
    Wrap(held: Dynamic),
  }

  record Marking {
    written: Dynamic;
  }

  fn loudness(wrapped: Wrapper) -> Nat {
    match wrapped {
      Wrap(Loud(level)) -> level,
      Wrap(Quiet) -> 0,
    }
  }

  fn opening(dynamics: List<Dynamic>) -> Nat {
    match dynamics {
      [] -> 0,
      [Loud(level), .. others] -> level,
      [Quiet, .. others] -> 0,
    }
  }

  fn marked(marking: Marking) -> Nat {
    match marking {
      Marking { written = Loud(level) } -> level,
      Marking { written = Quiet } -> 0,
    }
  }

  fn paired(both: Pair<Dynamic, Nat>) -> Nat {
    match both {
      (Loud(level), after) -> level,
      (Quiet, after) -> after,
    }
  }

  fn held(maybe: Option<Dynamic>) -> Nat {
    match maybe {
      Some(Loud(level)) -> level,
      Some(Quiet) -> 0,
      None -> 0,
    }
  }
}

library {
  data Clef { Treble, Bass, Alto, Tenor }
  data Form { Slurred, Grouped(played: Nat), Looped(times: Nat), Volta(pass: Nat) }

  fn low(written: Clef) -> Bool {
    match written {
      Bass | Tenor -> true,
      Treble | Alto -> false,
    }
  }

  fn plain(form: Form) -> Bool {
    match form {
      Slurred | Grouped(_) | Looped(_) | Volta(_) -> true,
    }
  }
}
