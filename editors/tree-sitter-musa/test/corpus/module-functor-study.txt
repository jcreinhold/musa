================================================================================
module-functor-study: a record of what a canon needs, and a function over it
================================================================================

// A bundle of facts that must travel together is a record; a value of it is
// what a `structure` was; a function over it is what a `template structure`
// was. The one thing this gives that a bag of parameters does not is that the
// facts travel *together*: a key, its collection, and how the two spell a
// degree are useless apart, and passing them one at a time is how they drift
// (docs/rules/language/01-surface.md §1.2).
import std::context;

// What a canon needs, gathered once. Fields, not members — a function is a
// value of arrow type, so one field form covers values, functions, and music
// alike.
record CanonMaterial {
    subject: EventTrack(WrittenTime);
    answer: EventTrack(WrittenTime) -> EventTrack(WrittenTime);
    canon: EventTrack(WrittenTime);
}

// The subject steps through whatever collection the context named. It reads
// the context through `TonalContext` and nothing else: a record's fields are
// its whole interface, so there is nothing else here to read by accident.
fn canon_subject(context: TonalContext) -> EventTrack(WrittenTime) {
    music {
        in scale context.collection {
            c5/4
            d5/4
            e5/4
            f5/4
        }
    }
}

// The answer, a fifth above. An ordinary music function, and a value of arrow
// type is what the record's second field holds.
fn canon_answer(line: EventTrack(WrittenTime)) -> EventTrack(WrittenTime) { transpose(P5, line) }

// The canon itself: subject against its own answer, delayed by whatever the
// caller asked for. The subject arrives as a parameter rather than as a local
// binding, which is what a `let` in a body would have been anyway.
fn canon_over(subject: EventTrack(WrittenTime), gap: Duration(WrittenTime)) -> CanonMaterial {
    CanonMaterial {
        subject = subject,
        answer = canon_answer,
        canon = together(subject, shift(gap, canon_answer(subject))),
    }
}

// The functor, which is a function.
fn canon(context: TonalContext, gap: Duration(WrittenTime)) -> CanonMaterial {
    canon_over(canon_subject(context), gap)
}

// Two values, two identities. Identity comes from the site the way it always
// did: two calls are two `let`s at two origins, and here the arguments differ
// as well.
let major_canon: CanonMaterial = canon(c_major, duration_of(1/1));

let minor_canon: CanonMaterial = canon(a_natural_minor, duration_of(2/1));

// One file is one piece, and this is it. What a piece reads out of a canon is
// an ordinary event track, and the piece never learns where it came from.
piece "Canon Duet" {
    meter 4/4;
    key c_major.tonic;

    score {
        part piano {
            voice upper {
                use major_canon.canon;
            }
            voice lower {
                use minor_canon.canon;
            }
        }
    }
}
