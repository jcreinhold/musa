piece "Scale Context" {
  import std::scale;
  let home: Key = key c minor;
  let collection: Scale = signature_scale(home);
  fn figure() -> Music {
    music {
      c5/8
      (c5 step 1)/8
      (c5 step down 2)/4
    }
  }
  score {
    part p {
      voice v {
        in scale c dorian {
          use figure();
        }
        in scale collection {
          c4/4
        }
      }
    }
  }
}
