================================================================================
loop-lengths
================================================================================

// A four-bar house pattern whose fills run "a few times" rather than a fixed
// number. `repeat 2 to 6 { … }` is the piece saying how long the figure may
// run and leaving the rest to the performance — the same sentence a DJ writes
// as "loop until the cue lands" and a folk singer writes as "as many verses as
// there are".
//
// The count is chosen once, at compile time, from the realization
// (`musa check examples/loop-lengths.musa --seed 42`). Everything below that
// choice is an ordinary exact repeat, which is why the event track gains nothing
// from this file: see docs/rules/events/11-realization.md.
piece "Loop Lengths" {
    composer "musa";

    tempo 1/4 = 124;
    meter 4/4;
    key a minor;

    score {
        part drums {
            voice kick {
                repeat 4 {
                    c2/4
                    c2/4
                    c2/4
                    c2/4
                }
                repeat 2 to 6 {
                    c2/4
                    c2/4
                    c2/8
                    c2/8
                    c2/4
                }
            }

            voice hats {
                repeat 4 {
                    rest/8
                    f#3/8
                    rest/8
                    f#3/8
                    rest/8
                    f#3/8
                    rest/8
                    f#3/8
                }
                repeat 2 to 6 {
                    f#3/8
                    f#3/8
                    f#3/8
                    f#3/8
                    f#3/8
                    f#3/8
                    f#3/8
                    f#3/8
                }
            }
        }

        part bass {
            voice line {
                repeat 4 {
                    a1/2
                    e2/4
                    g1/4
                }
                repeat 2 to 6 {
                    a1/4
                    a1/4
                    c2/4
                    e2/4
                }
            }
        }
    }
}
