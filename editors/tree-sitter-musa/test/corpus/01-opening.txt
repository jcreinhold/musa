// The album's first piece (docs/plan/prompts/36): what it shares it imports, and
// what changes over its length — the tempo, the loudness — is written where
// it changes rather than assumed constant.
piece "Opening" {
    import "../library/motifs.musa";
    import "../library/patches.musa";

    tempo 1/4 = 72;
    meter 4/4;
    key c major;

    score {
        part reeds {
            profile album;
            clef treble;

            voice lead {
                dynamic p;
                use rise();

                // The tempo is written where it changes, like the meter and
                // the key: a place in the piece, not a coordinate quoted from
                // somewhere else.
                tempo 1/4 = 108 "poco più mosso";

                crescendo to f {
                    use fall();
                }

                use rise();
                use fall();
            }
        }
    }

    studio {
        assign reeds -> reed;
        route reeds -> master;
    }
}
