// A piece in three keys. The first phrase is in F major, the second turns to
// D minor at the barline, and the last arrives in A major — which is what a
// key signature is for and what musa could not say until a `key` could be
// written where the music reaches it.
//
// Every change lands on a barline, because a key signature is printed at one.
// A clef change is the exception and has its own example.
piece "Modulation" {
    composer "musa";

    tempo 1/4 = 96;
    meter 4/4;
    key f major;

    score {
        part piano {
            clef treble;

            voice right {
                bar { f4 1/4; a4 1/4; c5 1/4; a4 1/4; }
                bar { bb4 1/4; a4 1/4; g4 1/4; f4 1/4; }

                key d minor;
                bar { d4 1/4; f4 1/4; a4 1/4; f4 1/4; }
                bar { g4 1/4; f4 1/4; e4 1/4; d4 1/4; }

                key a major;
                bar { a4 1/4; c#5 1/4; e5 1/4; c#5 1/4; }
                bar { a4 1/2; a4 1/2; }
            }
        }
    }
}
