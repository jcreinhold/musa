shuffle

// A twelve-bar blues, swung (docs/plan/prompts/69).
//
// Every eighth on this page is written straight, because that is how a blues
// is written and how a reader wants to read it. What makes it a shuffle is
// the word `swing` in the profile — the notated duration is not the performed
// duration (roadmap §2), and this is the first piece in the corpus where the
// two differ.
//
// The proof is in the exports rather than in the sound: the MEI, LilyPond and
// MusicXML of this piece contain even eighths, and the performance MIDI does
// not. An engraver that drew triplets here would be printing an
// interpretation nobody wrote down.
piece "Shuffle" {
    tempo 1/4 = 92;
    meter 4/4;
    key c major;

    // One bar of the ride pattern, twelve times. Written as material rather
    // than as a `repeat` among the voice's own items, because a repeat there
    // is a barline the other parts would have to share.
    fragment ride_bar {
        repeat 12 {
            d4/4
            d4/8
            d4/8
            d4/4
            d4/8
            d4/8
        }
    }

    performance {
        // Two thirds to the first of each pair: the ordinary long-short of
        // jazz and blues, and exact rather than a decimal that nearly is.
        profile band {
            groove swing {
                ratio = 2/3;
            }
            dynamic mf {
                amplitude = 0.6;
            }
            dynamic f {
                amplitude = 0.85;
            }
        }

        // The drummer keeps the same feel; the ride pattern is what carries
        // it, so the two profiles agree on purpose rather than by default.
        profile kit {
            groove swing {
                ratio = 2/3;
            }
            dynamic mf {
                amplitude = 0.55;
            }
        }
    }

    score {
        // The head: a riff on the tonic, answered on the subdominant and
        // resolved through the turnaround. Twelve bars, I-IV-I-V-IV-I.
        part lead {
            clef treble;
            profile band;

            voice tune {
                dynamic mf;

                // I — four bars
                c5/8
                eb5/8
                f5/8
                f#5/8
                g5/8
                f5/8
                eb5/8
                c5/8
                c5/8
                eb5/8
                f5/8
                f#5/8
                g5/4
                rest/4
                c5/8
                eb5/8
                f5/8
                f#5/8
                g5/8
                f5/8
                eb5/8
                c5/8
                bb4/4
                c5/4
                eb5/4
                f5/4

                // IV — two bars
                f5/8
                ab5/8
                bb5/8
                b5/8
                c6/8
                bb5/8
                ab5/8
                f5/8
                f5/8
                ab5/8
                bb5/8
                b5/8
                c6/4
                rest/4

                // I — two bars
                c5/8
                eb5/8
                f5/8
                f#5/8
                g5/8
                f5/8
                eb5/8
                c5/8
                c5/4
                eb5/4
                g5/4
                bb5/4

                // V — IV — I — turnaround
                dynamic f;
                g5/8
                b5/8
                d6/8
                f6/8
                g6/4
                rest/4
                f5/8
                a5/8
                c6/8
                eb6/8
                f6/4
                rest/4
                c5/8
                eb5/8
                f5/8
                f#5/8
                g5/8
                f5/8
                eb5/8
                c5/8
                g4/4
                bb4/4
                c5/2
            }
        }

        // The walking bass: quarters, which a swing leaves exactly where they
        // are — the warp fixes every beat boundary, so a walking line does
        // not shuffle just because the horn does.
        part bass {
            clef bass;
            profile band;

            voice walk {
                dynamic mf;

                c3/4
                e3/4
                g3/4
                a3/4
                c3/4
                e3/4
                g3/4
                bb3/4
                c3/4
                e3/4
                g3/4
                a3/4
                c3/4
                bb3/4
                a3/4
                ab3/4

                f3/4
                a3/4
                c4/4
                d4/4
                f3/4
                a3/4
                c4/4
                eb4/4

                c3/4
                e3/4
                g3/4
                a3/4
                c3/4
                e3/4
                g3/4
                bb3/4

                g3/4
                b3/4
                d4/4
                f4/4
                f3/4
                a3/4
                c4/4
                eb4/4
                c3/4
                e3/4
                g3/4
                a3/4
                g3/4
                f3/4
                e3/4
                c3/4
            }
        }

        // The ride: the pattern that states the shuffle, in the eighths the
        // groove actually moves.
        part drums {
            clef bass;
            profile kit;

            voice ride {
                dynamic mf;
                use ride_bar;
            }
        }
    }
}
