record Region {
  anchor: Nat;
  span: Nat;
}

record Pending<A> {
  read: A;
  region: Region;
}

enum Tying {
  Untied,
  TiedOn(Nat),
  Held { when: Nat; why: Text; },
}

enum Empty {
}

fn start() -> Region { Region { anchor = 0, span = 1 } }

fn untied() -> Tying { Tying::Untied }

fn shift(held: Pending<Nat>) -> Pending<Nat> { held with { region.anchor = 1 } }

fn width(held: Pending<Nat>) -> Nat { held.region.span }

fn tied(what: Tying) -> Nat {
  match what {
    Tying::Untied -> 0,
    Tying::TiedOn(n) -> n,
    Tying::Held { when = w } -> w,
  }
}

fn reading(held: Pending<Nat>) -> Nat {
  match held {
    Pending { read = r } -> r,
  }
}

fn anchored(where_: Region) -> Nat {
  match where_ {
    { anchor = a } -> a,
  }
}
