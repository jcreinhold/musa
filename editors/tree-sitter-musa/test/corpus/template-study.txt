================================================================================
template-study: reusable material as ordinary functions, and one piece
================================================================================

// A piece written once and made in whatever key it is wanted was a
// `template piece`. What was reusable about it is its *material* — a function
// of a key and a collection returning an event track — and that is ordinary
// source any number of pieces may import and call. The piece itself never was
// reusable, because a file is one piece however the piece got there
// (docs/rules/language/01-surface.md §6).
fn theme() -> EventTrack<WrittenTime> {
    music {
        c4/4
        d4/4
        e4/4
        f4/4
    }
}

// A voice that answers a subject through whatever transformation it is
// handed. Twice below: the same function, two voices, two identities.
fn answer(
    subject: EventTrack<WrittenTime>,
    transform: EventTrack<WrittenTime> -> EventTrack<WrittenTime>,
) -> EventTrack<WrittenTime> { transform(subject) }

// What was a template's parameter list is the file's lexical root. The study
// exists in whatever key these say.
let mode: Scale = scale g mixolydian;

let subject: EventTrack<WrittenTime> = theme();

piece "Study" {
    meter 4/4;
    key g major;

    score {
        part piano {
            voice right {
                in scale mode {
                    use subject;
                }
            }
            voice upper {
                use answer(
                    subject,
                    fn (line: EventTrack<WrittenTime>) -> EventTrack<WrittenTime> {
                        transpose(P8, line)
                    },
                );
            }
            voice higher {
                use answer(
                    subject,
                    fn (line: EventTrack<WrittenTime>) -> EventTrack<WrittenTime> {
                        transpose(P15, line)
                    },
                );
            }
        }
    }
}
