// Stockhausen's Klavierstück XI is nineteen fragments on one large sheet: the
// pianist looks away, looks back, and plays whichever fragment their eye lands
// on, until the sheet has been crossed three times. This file is that *shape* —
// the fragments are written for musa, not transcribed — and it is here because
// nineteen fragments have 19! orderings, which is the number that settled the
// argument in `docs/kernel/11-realization.md`: a finite `choose` cannot hold
// them, and a permutation costs nineteen `u32`s.
//
//     musa kernel examples/mobile.musa --seed 7
//
// The same seed is the same reading, every time; a different seed is a
// different one. That is law R1, and it is what makes a mobile a piece a
// recording can be made of rather than a lottery.
piece "Nineteen" {
    composer "musa, after the form of Klavierstück XI";

    tempo 1/4 = 92;
    meter 4/4;
    key c major;

    fragment group_one {
        b5 1/8;
        g4 1/8;
        af5 1/4;
        d5 1/2;
    }
    fragment group_two {
        bf4 1/2;
        cs5 1/2;
    }
    fragment group_three {
        fs5 1;
        c5 1;
    }
    fragment group_four {

        ef4 1/8;

    fragment group_twelve {
        ef4 1/8;

        ds5 1/8;

    }
    fragment group_thirteen {
        b5 1/8;
        g4 1/8;
        af5 1/4;
        d5 1/2;
    }
    fragment group_fourteen {
        bf4 1/2;
        cs5 1/2;
    }

    }
    fragment group_sixteen {

        e5 1/16;

    fragment group_seventeen {

        ds5 1/2;

        cs5 1/4;

        a4 1/4;

    }

    score {
