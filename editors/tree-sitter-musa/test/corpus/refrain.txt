piece "Refrain" {
    subtitle "a bar, named where it sounds";
    composer "musa";

    tempo quarter = 88;
    meter 3/4;
    key g major;

    score {
        part flute {
            clef treble;

            // A bar sounds where it is written and, with a name on the
            // front, can be played again from anywhere below it. `use` takes
            // no parentheses here because there is nothing to pass.
            voice tune {
                bar refrain { d5 1/4; b4 1/4; g4 1/4; }
                bar { a4 1/4; b4 1/4; c5 1/4; }
                use refrain;
                bar { e5 1/2; d5 1/4; }
                use refrain;
            }
        }

        part cello {
            clef bass;

            voice line {
                bar { g2 1/2; b2 1/4; }
                bar { d3 3/4; }
                bar { g2 1/2; b2 1/4; }
                bar { c3 1/2; a2 1/4; }
                bar { g2 3/4; }
            }
        }
    }
}
