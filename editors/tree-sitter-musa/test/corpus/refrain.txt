piece "Refrain" {
    subtitle "a bar, named where it sounds";
    composer "musa";

    tempo quarter = 88;
    meter 3/4;
    key g major;

    score {
        part flute {
            clef treble;

            // A bar sounds where it is written and, with a name on the
            // front, can be played again from anywhere below it. `use` takes
            // no parentheses here because there is nothing to pass.
            voice tune {
                bar refrain { d5/4 b4/4 g4/4 }
                | a4/4 b4/4 c5/4
                use refrain;
                | e5/2 d5/4
                use refrain;
            }
        }

        part cello {
            clef bass;

            voice line {
                | g2/2 b2/4
                | d3/2.
                | g2/2 b2/4
                | c3/2 a2/4
                | g2/2.
            }
        }
    }
}
