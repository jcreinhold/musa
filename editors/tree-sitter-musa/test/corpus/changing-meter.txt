================================================================================
changing-meter
================================================================================

// A Balkan-flavoured dance tune: the phrase is in 7/8 and the refrain answers
// in 4/4, which is ordinary in folk music and was inexpressible in musa until
// the meter learned to change where the music changes.
//
// The `meter` statements are written among the voice's own items, at the
// place they take effect — not as a coordinate, because a measure number is a
// position in bars and where the bars fall is what the meter decides.
piece "Changing Meter" {
    composer "traditional";
    arranger "musa";

    tempo 1/4 = 132;
    meter 7/8;
    key d minor;

    score {
        part fiddle {
            clef treble;

            voice tune {
                | d5/4 e5/4 f5/8 g5/8 a5/8
                | a5/4 g5/4 f5/4.

                meter 4/4;
                | d5/4 d5/4 a4/4 d5/4
                | f5/2 e5/2

                meter 7/8;
                | d5/8 e5/8 f5/8 g5/8 a5/8 g5/8 f5/8
                | e5/4 d5/4 d5/4.
            }
        }
    }
}
