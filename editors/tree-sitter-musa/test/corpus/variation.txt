// Variation, end to end (docs/plan/prompts/34): one theme, then the same theme
// stretched, backwards, mirrored, and, in the last bar, one occurrence
// specialized without touching the motif every other bar uses.
//
// Every transformation here is an elaboration-time function: the kernel gains
// nothing for any of them, and the score can still say where each note came
// from.
piece "Variation" {
    tempo 1/4 = 84;
    meter 4/4;
    key c major;

    motif theme() {
        c5/4
        e5/4
        g5/4
        e5/4
    }

    score {
        part piano {
            clef treble;

            voice lead {
                // The theme as written: four quarters, one bar.
                use theme();

                // Augmentation. Each quarter is written out as a half, so
                // the bar the theme filled once now takes two.
                stretch 2 {
                    use theme();
                }

                // The same four notes, last one first.
                retrograde {
                    use theme();
                }

                // Mirrored about the note the theme starts on: the rising
                // third becomes a falling one, spelled the way an engraver
                // would write it.
                invert around c5 {
                    use theme();
                }

                // One occurrence, one note different. The motif is untouched,
                // and so is every other bar that uses it.
                use theme() with {
                    note 3 = a5;
                }
            }
        }
    }
}
