// The expressive notation layer, end to end (docs/plan/prompts/27): tuplets,
// ties across a barline, slurs, dynamics, and articulations, including
// articulations written inside a motif, so provenance carries them too.
piece "Tuplet Fixture" {
    tempo 1/4 = 96;
    meter 4/4;
    key c major;

    motif turn(root: Pitch) {
        root/8 tenuto
        d5/8
    }

    score {
        part flute {
            clef treble;

            voice lead {
                dynamic mf;
                tuplet 3/2 {
                    c5/8
                    d5/8
                    e5/8
                }
                slur {
                    f5/8
                    g5/8
                }
                a5/4>
                g5/4 staccato ~
                g5/4
                dynamic p;
                use turn(a4);
                tuplet 3/2 {
                    c5/8
                    d5/8
                    e5/8
                }
                slur {
                    b4/8^
                    c5/8
                }
            }
        }

        part cello {
            clef bass;

            voice bass {
                dynamic p;
                [c3 g3]/2
                e3/4
                g3/4 ~
                g3/2
                slur {
                    a3/4 tenuto
                    g3/4
                }
            }
        }
    }
}
