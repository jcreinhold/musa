// Tempo changes, and the two things called tempo (docs/plan/prompts/72).
//
// A tempo marking is notation: it is printed, it stands at a place, and some
// markings are only words. The tempo *map* is a function from beats to
// seconds that the performance integrates. They are derived from the same
// declaration and they are not the same thing — which is why the third
// marking below prints and moves no clock at all.
//
// Three markings, of which two carry a metronome:
//
//   1/4 = 60                 at the top
//   1/4 = 120 "Doppio movimento"   at bar 3 — the music runs twice as fast
//   "Meno mosso"             at bar 5 — a word, and nothing for the engine
//
// So `--to lilypond` writes three `\tempo` commands, `--to musicxml` writes
// two `<sound tempo>` elements, and the MIDI file carries two set-tempo
// events. The page says more than the clock does, which is the ordinary case
// and the one a fused design cannot represent.
piece "Tempo Changes" {
    tempo 1/4 = 60;
    meter 4/4;
    key c major;

    score {
        part piano {
            clef treble;

            voice tune {
                dynamic mf;

                // Two bars at the opening tempo: a quarter note lasts a
                // second exactly, so this bar is four seconds long.
                c5/4
                d5/4
                e5/4
                f5/4

                g5/2
                e5/2

                // Twice as fast from here, and the word the composer wants
                // printed above the number.
                tempo 1/4 = 120 "Doppio movimento";
                g5/4
                a5/4
                b5/4
                c6/4

                b5/4
                a5/4
                g5/4
                f5/4

                // A word with no number. It prints; the clock does not move,
                // and the last two bars are still at 120. What "less" means
                // is the performer's, and — where musa answers it at all —
                // the profile's.
                tempo "Meno mosso";
                e5/2
                d5/2

                c5/1
            }
        }
    }
}
