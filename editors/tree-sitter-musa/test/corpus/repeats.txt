// A repeat is one statement with two projections: the page prints the body
// once between repeat barlines, and the performance plays it through twice.
// The endings are the part that differs — first time out, second time home.
piece "Two Ways Home" {
    tempo quarter = 108;
    meter 4/4;
    key g major;

    score {
        part flute {
            clef treble;

            voice tune {
                repeat 2 {
                    | d5/4 g5/4 f#5/4 g5/4
                    | a5/2 b5/2
                    ending 1 {
                        | a5/2 d5/2
                    }
                    ending 2 {
                        | g5/1
                    }
                }
            }
        }

        part cello {
            clef bass;

            voice line {
                repeat 2 {
                    | g2/2 b2/2
                    | d3/2 g2/2
                    ending 1 {
                        | c3/2 a2/2
                    }
                    ending 2 {
                        | g2/1
                    }
                }
            }
        }
    }
}
